% fig_unbounded_follower: why some MIPLIB-derived instances have no bilevel
% feasible point (recession-ray certificate, Meyer 1974 argument).
% The follower's region {y : D y >= b - C x} moves with the leader's choice x,
% but its recession cone {d : D d >= 0} does not.  If some d in that cone has
% f'd < 0, the follower's objective falls without bound along d for every x
% at which the follower is feasible, so no x has a best response.
% Schematic: three translates of one unbounded polyhedron (x1, x2, x3), each
% with the same direction d.
% Build: pdflatex fig_unbounded_follower.tex   (3.15 in x 2 in)

\pdfinfoomitdate=1 \pdftrailerid{} \pdfsuppressptexinfo=-1 % byte-identical rebuilds

\definecolor{follower}{HTML}{EB6834}
\definecolor{cert}{HTML}{1BAF7A}
\definecolor{ink}{HTML}{0B0B0B}
\definecolor{ink2}{HTML}{52514E}
\definecolor{muted}{HTML}{A9A8A2}
\definecolor{grid}{HTML}{E6E5E0}
\tikzset{
  every picture/.style={font=\fontsize{8}{9.6}\selectfont, color=ink, line cap=round, line join=round},
  small/.style={font=\fontsize{7}{8.4}\selectfont},
  ray/.style={cert, line width=1.3pt, -{Stealth[length=5pt,width=4.5pt]}},
}
\tikzfading[name=fade right, left color=transparent!0, right color=transparent!100]
\begin{document}
\begin{tikzpicture}
  % fixed page: 3.15 in x 2 in including the 2 pt border
  \useasboundingbox (0,0) rectangle (7.86,4.94);

  % axes of the follower's space
  \draw[ink2, line width=0.5pt, -{Stealth[length=3.5pt,width=3pt]}] (0.25,0.3) -- (7.7,0.3)
    node[anchor=north east, inner sep=1.5pt, text=ink2] {$y_1$};
  \draw[ink2, line width=0.5pt, -{Stealth[length=3.5pt,width=3pt]}] (0.25,0.3) -- (0.25,4.85)
    node[anchor=north west, inner sep=1.5pt, xshift=1pt, text=ink2] {$y_2$};

  % three leader choices (labels in the leader's blue): the follower's region
  % (orange) shifted (x1 front, x3 back);
  % corners A, B, lower edge slope 0.04, upper edge slope 0.25, cut at y1 = 7.4
  \foreach \dx/\dy/\op/\lab/\dash in {0.90/1.70/0.10/x_3/{densely dotted},
                                      0.45/0.85/0.16/x_2/{dashed},
                                      0.00/0.00/0.26/x_1/{solid}}{
    \begin{scope}[shift={(\dx,\dy)}]
      \coordinate (A) at (0.75,1.15);
      \coordinate (B) at (1.35,0.55);
      \coordinate (Bf) at (7.4-\dx,{0.55+0.04*(6.05-\dx)});
      \coordinate (Af) at (7.4-\dx,{1.15+0.25*(6.65-\dx)});
      \fill[follower, fill opacity=\op, path fading=fade right]
        (A) -- (B) -- (Bf) -- (Af) -- cycle;
      \draw[follower, line width=0.9pt, \dash, path fading=fade right]
        (Af) -- (A) -- (B) -- (Bf);
      \node[anchor=east, inner sep=1.5pt, text=leader] at (A) {$\lab$};
    \end{scope}
  }

  % the common direction d, drawn from inside each region
  \foreach \dx/\dy in {0.00/0.00, 0.45/0.85, 0.90/1.70}
    \draw[ray] (1.75+\dx,0.92+\dy) -- ++(1.6,0.22);
  \node[anchor=west, text=cert, inner sep=1pt] at (3.37,1.14) {$d$};

  % the message
  \node[anchor=north west, align=left, inner sep=0pt] at (0.95,4.82)
    {same direction $d$ for every $x$, and $f^\top d < 0$:\\
     $f^\top y \to -\infty$ along $d$, no best response,\\
     so no bilevel feasible point};
\end{tikzpicture}
\end{document}
